+"""Tests for retrying.py retry() functionality."""
+import pytest
+from retrying import retry
+
+
+def test_returns_first_result_without_retries():
+    """Test that first successful result is returned without extra calls."""
+    call_count = 0
+    def operation():
+        nonlocal call_count
+        call_count += 1
+        return 'success'
+
+    result = retry(operation, attempts=3)
+    assert result == 'success'
+    assert call_count == 1
+
+
+def test_retries_on_runtime_error_then_succeeds():
+    """Test that RuntimeError is retried and returns on success."""
+    call_count = 0
+    def operation():
+        nonlocal call_count
+        call_count += 1
+        if call_count < 3:
+            raise RuntimeError('temporary error')
+        return 'success'
+
+    result = retry(operation, attempts=3)
+    assert result == 'success'
+    assert call_count == 3
+
+
+def test_retries_all_attempts_then_raises_runtime_error():
+    """Test that final RuntimeError is propagated after all attempts."""
+    call_count = 0
+    def operation():
+        nonlocal call_count
+        call_count += 1
+        raise RuntimeError('permanently failed')
+
+    with pytest.raises(RuntimeError) as exc_info:
+        retry(operation, attempts=3)
+
+    assert call_count == 3
+    assert str(exc_info.value) == 'permanently failed'
+
+
